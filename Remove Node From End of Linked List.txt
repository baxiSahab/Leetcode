# --- Test Harness ---
def build(vals):
    dummy = ListNode(0)
    cur = dummy
    for v in vals:
        cur.next = ListNode(v)
        cur = cur.next
    return dummy.next

def to_list(head):
    result = []
    while head:
        result.append(head.val)
        head = head.next
    return result

tests = [
    ([1, 2, 3, 4, 5], 2, [1, 2, 3, 5]),   # remove 4th node
    ([1, 2, 3, 4, 5], 1, [1, 2, 3, 4]),   # remove last node
    ([1, 2, 3, 4, 5], 5, [2, 3, 4, 5]),   # remove head
    ([1],             1, []),              # single node
    ([1, 2],          1, [1]),             # two nodes, remove last
]

for vals, n, expected in tests:
    head = build(vals)
    result = to_list(removeNthFromEnd(head, n))
    status = "PASS" if result == expected else "FAIL"
    print(f"{status} | Input: {vals}, n={n} | Expected: {expected} | Got: {result}")
