class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def mergeTwoLists(list1: ListNode, list2: ListNode) -> ListNode:
    onne = list1
    twwo = list2
    dummy = ListNode(0)
    copy = dummy #saving head
    while onne or twwo:
        if onne == None:
            dummy.next = twwo
            return copy.next
        elif twwo == None:
            dummy.next = onne
            return copy.next
        # base cases ^^ to see if one list runs out
        if onne.val <= twwo.val:
            dummy.next = onne
            onne = onne.next
        else: 
            dummy.next = twwo
            twwo = twwo.next
        dummy = dummy.next
    return copy.next





# --- Test Harness ---
def build(vals):
    dummy = ListNode(0)
    cur = dummy
    for v in vals:
        cur.next = ListNode(v)
        cur = cur.next
    return dummy.next

def to_list(node):
    result = []
    while node:
        result.append(node.val)
        node = node.next
    return result

# Test 1: Basic merge
assert to_list(mergeTwoLists(build([1,2,4]), build([1,3,4]))) == [1,1,2,3,4,4]

# Test 2: One empty list
assert to_list(mergeTwoLists(build([]), build([1,2,3]))) == [1,2,3]

# Test 3: Both empty
assert to_list(mergeTwoLists(build([]), build([]))) == []

# Test 4: One list is a single node
assert to_list(mergeTwoLists(build([5]), build([1,2,6]))) == [1,2,5,6]

# Test 5: All elements from one list are smaller
assert to_list(mergeTwoLists(build([1,2,3]), build([4,5,6]))) == [1,2,3,4,5,6]

print("All tests passed!")
